   103→   103→function Toggle({ checked, onChange, disabled }: { checked: boolean; onChange: (v: boolean) => void; disabled?: boolean }) {
   104→   104→  return (
   105→   105→    <button
   106→   106→      type="button"
   107→   107→      role="switch"
   108→   108→      aria-checked={checked}
   109→   109→      disabled={disabled}
   110→   110→      onClick={() => onChange(!checked)}
   111→   111→      className={cn(
   112→   112→        'relative inline-flex h-5 w-9 shrink-0 cursor-pointer rounded-full border-2 border-transparent transition-colors duration-200 ease-in-out',
   113→   113→        'focus:outline-none focus-visible:ring-2 focus-visible:ring-blue-500 focus-visible:ring-offset-2',
   114→   114→        checked ? 'bg-blue-600' : 'bg-gray-300',
   115→   115→        disabled && 'opacity-50 cursor-not-allowed'
   116→   116→      )}
   117→   117→    >
   118→   118→      <span
   119→   119→        className={cn(
   120→   120→          'pointer-events-none inline-block h-4 w-4 rounded-full bg-white shadow ring-0 transition-transform duration-200 ease-in-out',
   121→   121→          checked ? 'translate-x-4' : 'translate-x-0'
   122→   122→        )}
   123→   123→      />
   124→   124→    </button>
   125→   125→  )
   126→   126→}
