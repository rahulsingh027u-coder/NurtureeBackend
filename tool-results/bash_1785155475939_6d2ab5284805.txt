   198	              {loading ? <Skeleton className="h-7 w-8" /> : <p className="text-2xl font-bold text-gray-900">{pendingVerifications.length}</p>}
   199	              <p className="text-xs text-gray-500">Pending Review</p>
   200	            </div>
   201	          </CardContent>
   202	        </Card>
   203	        <Card className="bg-white rounded-xl shadow-sm border-0">
   204	          <CardContent className="p-4 flex items-center gap-3">
   205	            <div className="w-10 h-10 rounded-full bg-blue-100 text-blue-600 flex items-center justify-center">
   206	              <ShieldCheck className="w-5 h-5" />
   207	            </div>
   208	            <div>
   209	              {loading ? <Skeleton className="h-7 w-8" /> : <p className="text-2xl font-bold text-gray-900">{allVerifications.filter((v) => v.status === 'approved').length}</p>}
   210	              <p className="text-xs text-gray-500">Approved</p>
   211	            </div>
   212	          </CardContent>
   213	        </Card>
   214	        <Card className="bg-white rounded-xl shadow-sm border-0">
   215	          <CardContent className="p-4 flex items-center gap-3">
   216	            <div className="w-10 h-10 rounded-full bg-red-100 text-red-600 flex items-center justify-center">
   217	              <ShieldCheck className="w-5 h-5" />
   218	            </div>
   219	            <div>
   220	              {loading ? <Skeleton className="h-7 w-8" /> : <p className="text-2xl font-bold text-gray-900">{allVerifications.filter((v) => v.status === 'rejected').length}</p>}
   221	              <p className="text-xs text-gray-500">Rejected</p>
   222	            </div>
   223	          </CardContent>
   224	        </Card>
   225	      </div>
   226	
   227	      {/* Tabs */}
   228	      <Tabs defaultValue="pending" className="space-y-4">
   229	        <div className="flex items-center justify-between">
   230	          <TabsList>
   231	            <TabsTrigger value="pending">Pending</TabsTrigger>
   232	            <TabsTrigger value="all">All</TabsTrigger>
   233	          </TabsList>
   234	          <Button
   235	            size="sm"
   236	            variant="outline"
   237	            className="text-xs border-blue-200 text-blue-700 hover:bg-blue-50 gap-1.5"
   238	            onClick={fetchVerifications}
   239	            disabled={loading}
   240	          >
   241	            <RefreshCw className={cn('w-3.5 h-3.5', loading && 'animate-spin')} />
   242	            Refresh
   243	          </Button>
   244	        </div>
   245	
   246	        <TabsContent value="pending">
   247	          <Card className="bg-white rounded-xl shadow-sm border-0">
   248	            <CardHeader className="pb-3">
   249	              <CardTitle className="text-base font-semibold text-gray-900">Pending Verifications</CardTitle>
   250	            </CardHeader>
   251	            <CardContent className="p-0">
   252	              {renderTable(pendingVerifications, true)}
   253	            </CardContent>
   254	          </Card>
   255	        </TabsContent>
   256	
   257	        <TabsContent value="all">
   258	          <Card className="bg-white rounded-xl shadow-sm border-0">
   259	            <CardHeader className="pb-3">
   260	              <CardTitle className="text-base font-semibold text-gray-900">All Verifications</CardTitle>
   261	            </CardHeader>
   262	            <CardContent className="p-0">
   263	              {renderTable(allVerifications, false)}
   264	            </CardContent>
   265	          </Card>
   266	        </TabsContent>
   267	      </Tabs>
   268	
   269	      {/* Approve / Reject Dialog */}
   270	      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   271	        <DialogContent className="sm:max-w-md">
   272	          <DialogHeader>
   273	            <DialogTitle className="flex items-center gap-2">
   274	              {action === 'approved' ? (
   275	                <ShieldCheck className="w-5 h-5 text-blue-600" />
   276	              ) : (
   277	                <AlertCircle className="w-5 h-5 text-red-600" />
   278	              )}
   279	              {action === 'approved' ? 'Approve Verification' : 'Reject Verification'}
   280	            </DialogTitle>
   281	          </DialogHeader>
   282	          {selectedVerification && (
   283	            <div className="space-y-4 py-2">
   284	              <div className="bg-gray-50 rounded-lg p-3 space-y-2">
   285	                <div className="flex items-center gap-2">
   286	                  <span className="text-xs text-gray-500 w-20">Name</span>
   287	                  <span className="text-sm font-medium text-gray-900">{selectedVerification.entityName}</span>
   288	                </div>
   289	                <div className="flex items-center gap-2">
   290	                  <span className="text-xs text-gray-500 w-20">Type</span>
   291	                  <Badge variant="secondary" className={cn('text-[11px] capitalize', entityTypeColor[selectedVerification.entityType])}>
   292	                    {selectedVerification.entityType}
   293	                  </Badge>
   294	                </div>
   295	                <div className="flex items-center gap-2">
   296	                  <span className="text-xs text-gray-500 w-20">Package</span>
   297	                  <span className="text-sm text-gray-700 capitalize">{selectedVerification.package}</span>
   298	                </div>
   299	              </div>
   300	              <div className="space-y-2">
   301	                <Label>Review Notes</Label>
   302	                <Textarea
   303	                  placeholder={action === 'approved' ? 'Optional approval notes...' : 'Reason for rejection...'}
   304	                  value={reviewNotes}
   305	                  onChange={(e) => setReviewNotes(e.target.value)}
   306	                  rows={3}
   307	                />
   308	              </div>
   309	            </div>
   310	          )}
   311	          <DialogFooter>
   312	            <Button variant="outline" onClick={() => setDialogOpen(false)}>Cancel</Button>
   313	            <Button
   314	              className={cn(
   315	                'text-white',
   316	                action === 'approved' ? 'bg-blue-600 hover:bg-blue-700' : 'bg-red-600 hover:bg-red-700'
   317	              )}
   318	              onClick={handleSubmitReview}
   319	              disabled={submitting}
   320	            >
   321	              {submitting && <Loader2 className="w-4 h-4 mr-2 animate-spin" />}
   322	              {action === 'approved' ? 'Approve' : 'Reject'}
   323	            </Button>
   324	          </DialogFooter>
   325	        </DialogContent>
   326	      </Dialog>
   327	    </div>
   328	  )
   329	}
