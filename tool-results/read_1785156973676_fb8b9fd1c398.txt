   102→/* ─── Toggle Switch Component ─── */
   103→function Toggle({ checked, onChange, disabled }: { checked: boolean; onChange: (v: boolean) => void; disabled?: boolean }) {
   104→  return (
   105→    <button
   106→      type="button"
   107→      role="switch"
   108→      aria-checked={checked}
   109→      disabled={disabled}
   110→      onClick={() => onChange(!checked)}
   111→      className={cn(
   112→        'relative inline-flex h-5 w-9 shrink-0 cursor-pointer rounded-full border-2 border-transparent transition-colors duration-200 ease-in-out',
   113→        'focus:outline-none focus-visible:ring-2 focus-visible:ring-blue-500 focus-visible:ring-offset-2',
   114→        checked ? 'bg-blue-600' : 'bg-gray-300',
   115→        disabled && 'opacity-50 cursor-not-allowed'
   116→      )}
   117→    >
   118→      <span
   119→        className={cn(
   120→          'pointer-events-none inline-block h-4 w-4 rounded-full bg-white shadow ring-0 transition-transform duration-200 ease-in-out',
   121→          checked ? 'translate-x-4' : 'translate-x-0'
   122→        )}
   123→      />
   124→    </button>
   125→  )
   126→}
